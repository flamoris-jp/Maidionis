cmake_minimum_required(VERSION 3.24)
project(Maidionis VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  message(FATAL_ERROR "The initial POSIX durability and numerical profile supports Linux x86_64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAIDIONIS_NUMERICAL "Build optional LibTorch numerical components" ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
file(GLOB_RECURSE MAIDIONIS_IDENTITY_SOURCES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
  "include/maidionis/*.h" "src/*.cpp" "contracts/*.schema.json" "education/python/src/maidionis_education/*.py")
list(APPEND MAIDIONIS_IDENTITY_SOURCES "CMakeLists.txt" "tests/tinybeat.py" "tests/fixture_data.py"
  "tests/tinybeat_composition.cpp" "tests/tinybeat_composition.h")
list(SORT MAIDIONIS_IDENTITY_SOURCES)
set(MAIDIONIS_BUILD_INPUT "maidionis-code-build-v1\n")
foreach(source IN LISTS MAIDIONIS_IDENTITY_SOURCES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_digest)
  string(APPEND MAIDIONIS_BUILD_INPUT "${source}\n${source_digest}\n")
endforeach()
string(SHA256 MAIDIONIS_BUILD_DIGEST "${MAIDIONIS_BUILD_INPUT}")
add_library(maidionis_contracts src/contracts/contracts.cpp)
target_include_directories(maidionis_contracts PUBLIC include ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(maidionis_contracts PUBLIC OpenSSL::Crypto)
target_compile_definitions(maidionis_contracts PRIVATE MAIDIONIS_SCHEMA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/contracts")
target_compile_definitions(maidionis_contracts PUBLIC MAIDIONIS_BUILD_DIGEST="${MAIDIONIS_BUILD_DIGEST}")
enable_testing()
add_executable(maidionis_contract_check src/app/contract_check.cpp)
target_link_libraries(maidionis_contract_check PRIVATE maidionis_contracts)
add_library(maidionis_storage src/data/storage.cpp)
target_link_libraries(maidionis_storage PUBLIC maidionis_contracts)
add_executable(maidionis_storage_tests tests/storage_test.cpp)
target_link_libraries(maidionis_storage_tests PRIVATE maidionis_storage)
add_test(NAME native_storage COMMAND maidionis_storage_tests)
add_library(maidionis_statistics src/evaluation/statistics.cpp)
target_include_directories(maidionis_statistics PUBLIC include)
add_executable(maidionis_statistics_tests tests/statistics_test.cpp)
target_link_libraries(maidionis_statistics_tests PRIVATE maidionis_statistics)
add_test(NAME native_statistics COMMAND maidionis_statistics_tests)
if(MAIDIONIS_NUMERICAL)
  find_package(Torch 2.5.1 EXACT REQUIRED)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
  add_library(maidionis_core src/model/model.cpp src/training/training.cpp src/artifact/artifact.cpp)
  target_link_libraries(maidionis_core PUBLIC maidionis_storage ${TORCH_LIBRARIES})
  string(TOUPPER "${CMAKE_BUILD_TYPE}" MAIDIONIS_BUILD_TYPE_UPPER)
  target_compile_definitions(maidionis_core PRIVATE MAIDIONIS_BUILD_TYPE="${CMAKE_BUILD_TYPE}"
    MAIDIONIS_COMPILER_FLAGS="${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${MAIDIONIS_BUILD_TYPE_UPPER}}")
  add_executable(maidionis_model_tests tests/model_test.cpp)
  target_link_libraries(maidionis_model_tests PRIVATE maidionis_core)
  add_test(NAME native_model COMMAND maidionis_model_tests)
  add_library(maidionis_tinybeat_composition STATIC tests/tinybeat_composition.cpp)
  target_link_libraries(maidionis_tinybeat_composition PUBLIC maidionis_core)
  target_include_directories(maidionis_tinybeat_composition PUBLIC tests)
  add_executable(maidionis_tinybeat_driver tests/tinybeat_driver.cpp)
  target_link_libraries(maidionis_tinybeat_driver PRIVATE maidionis_tinybeat_composition)
endif()
