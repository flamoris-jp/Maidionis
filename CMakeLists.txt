cmake_minimum_required(VERSION 3.24)
project(Maidionis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAIDIONIS_NUMERICAL "Build optional LibTorch numerical components" ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(maidionis_contracts src/contracts/contracts.cpp)
target_include_directories(maidionis_contracts PUBLIC include ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(maidionis_contracts PUBLIC OpenSSL::Crypto)
target_compile_definitions(maidionis_contracts PRIVATE MAIDIONIS_SCHEMA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/contracts")
enable_testing()
add_executable(maidionis_contract_check src/app/contract_check.cpp)
target_link_libraries(maidionis_contract_check PRIVATE maidionis_contracts)
