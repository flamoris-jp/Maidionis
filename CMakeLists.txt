cmake_minimum_required(VERSION 3.24)
project(Maidionis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAIDIONIS_NUMERICAL "Build optional LibTorch numerical components" ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(maidionis_contracts src/contracts/contracts.cpp)
target_include_directories(maidionis_contracts PUBLIC include ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(maidionis_contracts PUBLIC OpenSSL::Crypto)
target_compile_definitions(maidionis_contracts PRIVATE MAIDIONIS_SCHEMA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/contracts")
enable_testing()
add_executable(maidionis_contract_check src/app/contract_check.cpp)
target_link_libraries(maidionis_contract_check PRIVATE maidionis_contracts)
add_library(maidionis_storage src/data/storage.cpp)
target_link_libraries(maidionis_storage PUBLIC maidionis_contracts)
add_executable(maidionis_storage_tests tests/storage_test.cpp)
target_link_libraries(maidionis_storage_tests PRIVATE maidionis_storage)
add_test(NAME native_storage COMMAND maidionis_storage_tests)
if(MAIDIONIS_NUMERICAL)
  find_package(Torch 2.5.1 EXACT REQUIRED)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
  add_library(maidionis_core src/model/model.cpp src/training/training.cpp)
  target_link_libraries(maidionis_core PUBLIC maidionis_storage ${TORCH_LIBRARIES})
  add_executable(maidionis_model_tests tests/model_test.cpp)
  target_link_libraries(maidionis_model_tests PRIVATE maidionis_core)
  add_test(NAME native_model COMMAND maidionis_model_tests)
  add_library(maidionis_tinybeat_composition STATIC tests/tinybeat_composition.cpp)
  target_link_libraries(maidionis_tinybeat_composition PUBLIC maidionis_core)
  target_include_directories(maidionis_tinybeat_composition PUBLIC tests)
  add_executable(maidionis_tinybeat_driver tests/tinybeat_driver.cpp)
  target_link_libraries(maidionis_tinybeat_driver PRIVATE maidionis_tinybeat_composition)
endif()
